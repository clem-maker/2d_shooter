var velocity : Vector2
var SPEED : float
func _ready() -> void:
	gravity_scale = 0

func _process(delta: float) -> void:
	if not Global.target_vec == Vector2.ZERO:
		#linear_velocity = Global.target_vec
		velocity = Global.target_vec
		var vector_length:float = sqrt(Global.target_vec.x**2 + Global.target_vec.y **2)
		#print("vector lenght", vector_length)
		SPEED = vector_length
		
	
		position += Global.target_vec *SPEED *delta /10
	delta = delta
